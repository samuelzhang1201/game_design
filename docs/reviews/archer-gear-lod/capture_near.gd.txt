extends SceneTree
func _initialize() -> void:
	call_deferred("shoot")
func shoot() -> void:
	root.size = Vector2i(1000, 1000)
	var world := Node3D.new()
	root.add_child(world)
	var environment_node := WorldEnvironment.new()
	var environment := Environment.new()
	environment.background_mode = Environment.BG_COLOR
	environment.background_color = Color(0.20, 0.20, 0.20)
	environment.ambient_light_source = Environment.AMBIENT_SOURCE_COLOR
	environment.ambient_light_color = Color.WHITE
	environment.ambient_light_energy = 0.65
	environment_node.environment = environment
	world.add_child(environment_node)
	var sun := DirectionalLight3D.new()
	sun.rotation_degrees = Vector3(-45, -30, 0)
	sun.light_energy = 1.5
	world.add_child(sun)
	var camera := Camera3D.new()
	camera.projection = Camera3D.PROJECTION_ORTHOGONAL
	camera.size = 2.7
	world.add_child(camera)
	var packed: PackedScene = load("res://scenes/models/units/archers.tscn")
	for variant: String in ["before", "conservative", "proposed"]:
		var model := packed.instantiate()
		world.add_child(model)
		for child: Node in model.find_children("*", "MeshInstance3D", true, false):
			child.lod_bias = 100000.0
		if variant != "before":
			for part: String in ["bow", "arrow", "quiver"]:
				var directory := "/tmp/archer_gear_final/" + variant
				var instance := model.find_child("archers_" + part, true, false) as MeshInstance3D
				instance.mesh = load(directory.path_join("archers_" + part + ".res"))
				var material := instance.get_surface_override_material(0).duplicate() as StandardMaterial3D
				for channel: String in ["base", "mr", "normal"]:
					if part == "quiver" and channel == "mr":
						continue
					var pixels := Image.load_from_file(directory.path_join(part + "_" + channel + ".jpg"))
					assert(pixels != null)
					pixels.generate_mipmaps()
					var texture := ImageTexture.create_from_image(pixels)
					if channel == "base": material.albedo_texture = texture
					elif channel == "normal": material.normal_texture = texture
					else:
						material.roughness_texture = texture
						material.metallic_texture = texture
				instance.set_surface_override_material(0, material)
		var player := Map3DUnits.find_anim(model)
		for clip: String in ["STANDBY", "MARCH", "SPRINT", "DRAW", "GARRISON", "DEATH"]:
			player.play(Map3DUnits.clip_named(player, clip))
			player.seek(player.current_animation_length * 0.5, true)
			player.pause()
			for angle: String in ["front", "back"]:
				camera.position = Vector3(1.8, 1.8, -3) if angle == "front" else Vector3(-1.8, 1.8, 3)
				camera.look_at(Vector3(0, 1, 0))
				await process_frame
				await RenderingServer.frame_post_draw
				root.get_texture().get_image().save_png("/tmp/unit_detail_review/final_" + variant + "_" + clip + "_" + angle + ".png")
		model.queue_free()
		await process_frame
	quit()
