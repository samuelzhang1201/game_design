**Goal:** Import the Azgaar "Beria" exports into game-data CSVs plus a hit-test bitmap, and add a pan/zoom world-map overlay with clickable kingdoms.

**Architecture:** A pure-stdlib Python importer (`tools/import_azgaar.py`) turns the Azgaar full JSON into `kingdoms.csv`/`provinces.csv`/`burgs.csv` and rasterizes cell polygons into a state-index PNG (state id encoded in R/G channels). `DataRepository` loads the CSVs; a new `MapScreen` overlay (toggled by the bottom bar's MAP tab) draws the map with pan/zoom and resolves clicks through the index bitmap via a tiny engine-pure `MapIndex` helper.

**Tech Stack:** Python 3 stdlib only (json/zlib/struct/csv — Pillow is NOT installed on this machine; do not import it), Godot 4 typed GDScript, GUT.

**Spec:** `docs/superpowers/specs/2026-09-03-beria-map-integration-design.md`

### Azgaar JSON facts (verified against `assets/maps/Beria Full 2026-09-03-20-04.json`, v1.150.0)

- `info.width` = 2560, `info.height` = 929 — 1:1 with the PNG's pixels.
- `pack.cells`: list of dicts. Relevant keys: `i` (id), `v` (vertex-id list), `p` ([x, y] center), `h` (height), `state`, `province`, `pop` (rural population points), `area`. 6045 cells; 4156 have `state > 0`.
- `pack.vertices`: list of dicts; `p` = [x, y] floats.
- `pack.states`: 17 entries, ids 0–16. State 0 is `Neutrals` (exclude). Keys: `i`, `name`, `capital` (burg id), `center` (cell id), `neighbors` (state-id list, may contain 0), `color` (`#rrggbb`), `rural`, `urban` (population points), `area`.
- `pack.provinces`: index 0 is the integer `0`, not a dict — skip non-dict entries (same for burgs). Province keys: `i`, `state`, `burg` (capital burg id, may be 0 = none), `name`. Provinces carry NO area/population — aggregate from cells (`cell.province == i`).
- `pack.burgs`: burg keys: `i`, `name`, `state`, `cell`, `x`, `y`, `capital` (0/1), `port` (0/1), `population` (points). A burg's province = `cells[burg.cell].province`. Skip entries with `removed` truthy (0 in this export, but future re-exports may have them — same for provinces/states).
- `settings.populationRate` = 1000: people = points × rate. Report populations as `int(round(points * rate))`.

```python
#!/usr/bin/env python3
"""Import an Azgaar Fantasy Map Generator full export into game data.

Reads the newest "*Full*.json" and newest map render PNG in assets/maps/
and writes (all committed to git; re-run after every re-export):

  data/kingdoms.csv    one row per state (state 0 "Neutrals" excluded)
  data/provinces.csv   one row per province
  data/burgs.csv       one row per burg
  assets/maps/world_map.png          byte copy under a stable name
  assets/maps/world_state_index.png  hit-test bitmap: each land cell's
      polygon filled with R = state_id & 0xFF, G = state_id >> 8, B = 0;
      untouched pixels stay white (255,255,255) = water / no state

Stdlib only (no Pillow): the PNG is written with zlib + struct, polygons
are filled with an even-odd scanline rasterizer. Deterministic: same
inputs => byte-identical outputs. Hand-edits to the generated CSVs are
overwritten; lasting changes belong in Azgaar.

Usage:  python3 tools/import_azgaar.py [--json PATH] [--png PATH]
"""
from __future__ import annotations

import argparse
import csv
import json
import re
import shutil
import struct
import sys
import zlib
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
MAPS = ROOT / "assets" / "maps"
DATA = ROOT / "data"
FICTION_KINGDOMS = {"averlain", "bentleigh", "doncaster", "empire"}
GENERATED_PNGS = {"world_map.png", "world_state_index.png"}


def slugify(name: str) -> str:
    slug = re.sub(r"[^a-z0-9]+", "_", name.lower()).strip("_")
    return slug or "unnamed"


class SlugTable:
    """Hands out collision-free slugs (foo, foo_2, foo_3, ...)."""

    def __init__(self) -> None:
        self._used: dict[str, int] = {}

    def take(self, name: str) -> str:
        base = slugify(name)
        n = self._used.get(base, 0) + 1
        self._used[base] = n
        return base if n == 1 else f"{base}_{n}"


def newest(pattern_hits: list[Path], what: str) -> Path:
    if not pattern_hits:
        sys.exit(f"error: no {what} found in {MAPS}")
    return max(pattern_hits, key=lambda p: p.stat().st_mtime)


def active(entries: list) -> list[dict]:
    """Azgaar lists start with a literal 0 and may flag removed rows."""
    return [e for e in entries if isinstance(e, dict) and not e.get("removed")]


def png_dimensions(path: Path) -> tuple[int, int]:
    head = path.read_bytes()[:33]
    if head[:8] != b"\x89PNG\r\n\x1a\n" or head[12:16] != b"IHDR":
        sys.exit(f"error: {path} is not a PNG")
    w, h = struct.unpack(">II", head[16:24])
    return w, h


def write_png_rgb(path: Path, width: int, height: int, buf: bytearray) -> None:
    """Minimal RGB8 PNG writer: buf is width*height*3 bytes, row-major."""

    def chunk(tag: bytes, payload: bytes) -> bytes:
        return (struct.pack(">I", len(payload)) + tag + payload
                + struct.pack(">I", zlib.crc32(tag + payload) & 0xFFFFFFFF))

    stride = width * 3
    raw = b"".join(
        b"\x00" + bytes(buf[y * stride:(y + 1) * stride]) for y in range(height))
    ihdr = struct.pack(">IIBBBBB", width, height, 8, 2, 0, 0, 0)
    path.write_bytes(b"\x89PNG\r\n\x1a\n" + chunk(b"IHDR", ihdr)
                     + chunk(b"IDAT", zlib.compress(raw, 9)) + chunk(b"IEND", b""))


def fill_polygon(buf: bytearray, width: int, height: int,
                 pts: list[tuple[float, float]], state_id: int) -> None:
    """Even-odd scanline fill sampling pixel centers (x+0.5, y+0.5)."""
    r, g = state_id & 0xFF, (state_id >> 8) & 0xFF
    ys = [p[1] for p in pts]
    y_lo = max(0, int(min(ys)))
    y_hi = min(height - 1, int(max(ys)) + 1)
    n = len(pts)
    for y in range(y_lo, y_hi + 1):
        yc = y + 0.5
        xs: list[float] = []
        for i in range(n):
            x1, y1 = pts[i]
            x2, y2 = pts[(i + 1) % n]
            if (y1 <= yc < y2) or (y2 <= yc < y1):
                xs.append(x1 + (yc - y1) / (y2 - y1) * (x2 - x1))
        xs.sort()
        for j in range(0, len(xs) - 1, 2):
            x_lo = max(0, int(xs[j] + 0.5))
            x_hi = min(width - 1, int(xs[j + 1] - 0.5))
            for x in range(x_lo, x_hi + 1):
                o = (y * width + x) * 3
                buf[o] = r
                buf[o + 1] = g
                buf[o + 2] = 0


def main() -> None:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--json", type=Path, default=None)
    ap.add_argument("--png", type=Path, default=None)
    args = ap.parse_args()

    json_path = args.json or newest(
        sorted(MAPS.glob("*Full*.json")), "Azgaar full export (*Full*.json)")
    png_path = args.png or newest(
        [p for p in MAPS.glob("*.png") if p.name not in GENERATED_PNGS],
        "map render PNG")
    print(f"json: {json_path.name}\npng:  {png_path.name}")

    world = json.loads(json_path.read_text())
    width, height = world["info"]["width"], world["info"]["height"]
    png_w, png_h = png_dimensions(png_path)
    if (png_w, png_h) != (width, height):
        sys.exit(f"error: JSON says {width}x{height} but PNG is {png_w}x{png_h}")

    pack = world["pack"]
    rate = float(world["settings"].get("populationRate", 1000))
    cells: list[dict] = pack["cells"]
    vertices: list[dict] = pack["vertices"]
    states = [s for s in active(pack["states"]) if s["i"] != 0]
    provinces = active(pack["provinces"])
    burgs = active(pack["burgs"])

    # --- codes ------------------------------------------------------------
    state_code = {s["i"]: slugify(s["name"]) for s in states}
    if len(set(state_code.values())) != len(state_code):
        sys.exit("error: two states slugify to the same kingdom_code")
    prov_slugs, burg_slugs = SlugTable(), SlugTable()
    prov_code = {p["i"]: prov_slugs.take(p["name"]) for p in provinces}
    burg_code = {b["i"]: burg_slugs.take(b["name"]) for b in burgs}

    # --- referential checks ----------------------------------------------
    for p in provinces:
        if p["state"] not in state_code:
            sys.exit(f"error: province {p['i']} references missing state {p['state']}")
    burg_province: dict[int, int] = {}
    for b in burgs:
        if b["state"] not in state_code:
            sys.exit(f"error: burg {b['i']} references missing state {b['state']}")
        prov_id = cells[b["cell"]].get("province", 0)
        if prov_id not in prov_code:
            sys.exit(f"error: burg {b['i']} cell {b['cell']} has no province")
        burg_province[b["i"]] = prov_id

    # --- aggregate cells per province ------------------------------------
    prov_area: dict[int, float] = {i: 0.0 for i in prov_code}
    prov_pop: dict[int, float] = {i: 0.0 for i in prov_code}
    for c in cells:
        pid = c.get("province", 0)
        if pid in prov_area:
            prov_area[pid] += c.get("area", 0)
            prov_pop[pid] += c.get("pop", 0)
    for b in burgs:  # urban population on top of the cells' rural points
        prov_pop[burg_province[b["i"]]] += b.get("population", 0)

    # --- kingdoms.csv -----------------------------------------------------
    DATA.mkdir(exist_ok=True)
    with (DATA / "kingdoms.csv").open("w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["kingdom_code", "state_id", "label", "capital_burg",
                    "neighbor_codes", "color", "area", "population",
                    "center_x", "center_y"])
        for s in sorted(states, key=lambda s: s["i"]):
            center = cells[s["center"]]["p"]
            neighbors = "|".join(state_code[n] for n in s.get("neighbors", [])
                                 if n in state_code)
            w.writerow([state_code[s["i"]], s["i"], s["name"],
                        burg_code.get(s.get("capital", 0), ""),
                        neighbors, s.get("color", ""), int(s.get("area", 0)),
                        int(round((s.get("rural", 0) + s.get("urban", 0)) * rate)),
                        round(center[0], 2), round(center[1], 2)])

    # --- provinces.csv ----------------------------------------------------
    with (DATA / "provinces.csv").open("w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["province_code", "label", "kingdom_code", "capital_burg",
                    "area", "population"])
        for p in sorted(provinces, key=lambda p: p["i"]):
            w.writerow([prov_code[p["i"]], p["name"], state_code[p["state"]],
                        burg_code.get(p.get("burg", 0), ""),
                        int(prov_area[p["i"]]),
                        int(round(prov_pop[p["i"]] * rate))])

    # --- burgs.csv --------------------------------------------------------
    with (DATA / "burgs.csv").open("w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["burg_code", "label", "province_code", "kingdom_code",
                    "population", "is_capital", "is_port", "x", "y"])
        for b in sorted(burgs, key=lambda b: b["i"]):
            w.writerow([burg_code[b["i"]], b["name"],
                        prov_code[burg_province[b["i"]]], state_code[b["state"]],
                        int(round(b.get("population", 0) * rate)),
                        1 if b.get("capital") else 0, 1 if b.get("port") else 0,
                        round(b["x"], 2), round(b["y"], 2)])

    # --- bitmaps ----------------------------------------------------------
    shutil.copyfile(png_path, MAPS / "world_map.png")
    buf = bytearray(b"\xff" * (width * height * 3))
    for c in cells:
        sid = c.get("state", 0)
        if sid in state_code:
            pts = [tuple(vertices[v]["p"]) for v in c["v"]]
            fill_polygon(buf, width, height, pts, sid)
    write_png_rgb(MAPS / "world_state_index.png", width, height, buf)

    # --- report -----------------------------------------------------------
    missing = sorted(FICTION_KINGDOMS - set(state_code.values()))
    if missing:
        print(f"WARNING: fiction kingdoms not on the map yet: {', '.join(missing)}"
              "\n         (expected until states are renamed in Azgaar and re-exported)")
    print(f"wrote {len(states)} kingdoms, {len(provinces)} provinces, "
          f"{len(burgs)} burgs, world_map.png, world_state_index.png")


if __name__ == "__main__":
    main()
```

`scripts/engine/map_index.gd`:

```gdscript
class_name MapIndex
extends RefCounted
## Decodes the state-index bitmap generated by tools/import_azgaar.py:
## each land pixel carries its Azgaar state id as R | (G << 8), B = 0;
## pure white is water / no state. -1 means "nothing here".

static func decode(c: Color) -> int:
	if c.r8 == 255 and c.g8 == 255 and c.b8 == 255:
		return -1
	return c.r8 | (c.g8 << 8)

static func state_at(img: Image, px: Vector2i) -> int:
	if img == null or px.x < 0 or px.y < 0 \
			or px.x >= img.get_width() or px.y >= img.get_height():
		return -1
	return decode(img.get_pixelv(px))
```

`scripts/ui/map_screen.gd`:

```gdscript
class_name MapScreen
extends Control
## Full-screen world-map overlay toggled by the bottom bar's MAP tab (user
## ruling 2026-09-03, supersedes 2026-08-30's "MAP closes overlays"): a
## pan/zoom view of assets/maps/world_map.png with clickable kingdoms,
## resolved through the world_state_index.png bitmap (tools/import_azgaar.py)
## via MapIndex. View-only: reads Game.data, never mutates game state.

const MAP_TEXTURE_PATH := "res://assets/maps/world_map.png"
const INDEX_TEXTURE_PATH := "res://assets/maps/world_state_index.png"
const ZOOM_MIN := 0.5
const ZOOM_MAX := 4.0
const ZOOM_STEP := 1.2
## A press/release pair further apart than this is a drag, not a click.
const CLICK_SLOP := 5.0
const PANEL_WIDTH := 200.0

var _map_texture: Texture2D
var _index_image: Image
var _by_state_id: Dictionary = {}
var _view: Control
var _panel_title: Label
var _panel_body: Label
var _offset := Vector2.ZERO
var _zoom := 1.0
var _fitted := false
var _dragging := false
var _press_pos := Vector2.ZERO

func _ready() -> void:
	set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	mouse_filter = Control.MOUSE_FILTER_STOP
	var dim := ColorRect.new()
	dim.color = UiPalette.DIM_OVERLAY
	dim.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	dim.mouse_filter = Control.MOUSE_FILTER_IGNORE
	add_child(dim)
	if ResourceLoader.exists(MAP_TEXTURE_PATH):
		_map_texture = load(MAP_TEXTURE_PATH)
	if ResourceLoader.exists(INDEX_TEXTURE_PATH):
		var idx: Texture2D = load(INDEX_TEXTURE_PATH)
		_index_image = idx.get_image()
	for code: String in Game.data.kingdoms:
		var row: Dictionary = Game.data.kingdoms[code]
		_by_state_id[int(row["state_id"])] = row
	_build_view()
	_build_panel()
	hide()

func _build_view() -> void:
	_view = Control.new()
	_view.name = "MapView"
	_view.clip_contents = true
	_view.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	_view.offset_right = -PANEL_WIDTH
	_view.mouse_filter = Control.MOUSE_FILTER_STOP
	_view.draw.connect(_draw_view)
	_view.gui_input.connect(_on_view_input)
	_view.resized.connect(func() -> void: _fitted = false; _view.queue_redraw())
	add_child(_view)

func _build_panel() -> void:
	var panel := PanelContainer.new()
	panel.name = "InfoPanel"
	panel.add_theme_stylebox_override("panel",
		UiPalette.panel_style(UiPalette.GOLD, 1, 8))
	panel.set_anchors_and_offsets_preset(Control.PRESET_RIGHT_WIDE)
	panel.offset_left = -PANEL_WIDTH
	add_child(panel)
	var col := VBoxContainer.new()
	col.add_theme_constant_override("separation", 5)
	panel.add_child(col)
	_panel_title = Label.new()
	_panel_title.name = "PanelTitle"
	_panel_title.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	UiPalette.style_header(_panel_title, 11)
	col.add_child(_panel_title)
	col.add_child(HSeparator.new())
	_panel_body = Label.new()
	_panel_body.name = "PanelBody"
	_panel_body.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	_panel_body.size_flags_vertical = Control.SIZE_EXPAND_FILL
	_panel_body.add_theme_color_override("font_color", UiPalette.TEXT_MUTED)
	col.add_child(_panel_body)
	var close := Button.new()
	close.name = "CloseButton"
	close.text = "CLOSE"
	close.focus_mode = Control.FOCUS_NONE
	close.pressed.connect(func() -> void: hide())
	col.add_child(close)
	_clear_panel()

func _draw_view() -> void:
	if _map_texture == null:
		_view.draw_string(get_theme_default_font(),
			Vector2(16, 24), "No map imported — run tools/import_azgaar.py")
		return
	if not _fitted:
		_fit_view()
	_view.draw_texture_rect(_map_texture,
		Rect2(_offset, _map_texture.get_size() * _zoom), false)

## Initial framing: fit the map's height, centered horizontally.
func _fit_view() -> void:
	var s := _view.size
	if s.y <= 0.0:
		return
	_zoom = clampf(s.y / _map_texture.get_height(), ZOOM_MIN, ZOOM_MAX)
	_offset = Vector2((s.x - _map_texture.get_width() * _zoom) * 0.5, 0.0)
	_clamp_offset()
	_fitted = true

func _on_view_input(ev: InputEvent) -> void:
	if _map_texture == null:
		return
	if ev is InputEventMouseButton:
		var mb := ev as InputEventMouseButton
		if mb.button_index == MOUSE_BUTTON_WHEEL_UP and mb.pressed:
			_zoom_at(mb.position, ZOOM_STEP)
		elif mb.button_index == MOUSE_BUTTON_WHEEL_DOWN and mb.pressed:
			_zoom_at(mb.position, 1.0 / ZOOM_STEP)
		elif mb.button_index == MOUSE_BUTTON_LEFT:
			if mb.pressed:
				_dragging = false
				_press_pos = mb.position
			elif not _dragging:
				_select_at(mb.position)
	elif ev is InputEventMouseMotion:
		var mm := ev as InputEventMouseMotion
		if mm.button_mask & MOUSE_BUTTON_MASK_LEFT:
			if _press_pos.distance_to(mm.position) > CLICK_SLOP:
				_dragging = true
			if _dragging:
				_offset += mm.relative
				_clamp_offset()
				_view.queue_redraw()

func _zoom_at(pos: Vector2, factor: float) -> void:
	var before := (pos - _offset) / _zoom
	_zoom = clampf(_zoom * factor, ZOOM_MIN, ZOOM_MAX)
	_offset = pos - before * _zoom
	_clamp_offset()
	_view.queue_redraw()

func _clamp_offset() -> void:
	var scaled := _map_texture.get_size() * _zoom
	_offset.x = _clamp_axis(_offset.x, _view.size.x, scaled.x)
	_offset.y = _clamp_axis(_offset.y, _view.size.y, scaled.y)

func _clamp_axis(o: float, view_len: float, map_len: float) -> float:
	if map_len <= view_len:
		return clampf(o, 0.0, view_len - map_len)
	return clampf(o, view_len - map_len, 0.0)

func _select_at(pos: Vector2) -> void:
	var map_px := ((pos - _offset) / _zoom).floor()
	select_state(MapIndex.state_at(_index_image, Vector2i(map_px)))

## Public for tests: show the kingdom with this Azgaar state id, or clear
## the panel for -1 / unknown ids (stale bitmap vs. CSVs degrades silently).
func select_state(sid: int) -> void:
	var row: Dictionary = _by_state_id.get(sid, {})
	if row.is_empty():
		_clear_panel()
	else:
		_show_kingdom(row)

func _clear_panel() -> void:
	_panel_title.text = ""
	_panel_body.text = "Click a kingdom."

func _show_kingdom(row: Dictionary) -> void:
	var code := String(row["kingdom_code"])
	var provinces := 0
	for p: Dictionary in Game.data.provinces.values():
		if String(p["kingdom_code"]) == code:
			provinces += 1
	var towns := 0
	for b: Dictionary in Game.data.burgs.values():
		if String(b["kingdom_code"]) == code:
			towns += 1
	var capital: Dictionary = Game.data.burgs.get(String(row["capital_burg"]), {})
	var neighbors: Array[String] = []
	for n: String in String(row["neighbor_codes"]).split("|", false):
		neighbors.append(String(Game.data.kingdoms.get(n, {}).get("label", n)))
	var court: Array[String] = []
	for role: Dictionary in Game.data.roles:
		if String(role.get("kingdom", "")) == code:
			court.append(String(role["display_name"]))
	_panel_title.text = String(row["label"])
	var lines: Array[String] = [
		"Capital: %s" % capital.get("label", "—"),
		"Provinces: %d" % provinces,
		"Towns: %d" % towns,
		"Population: %s" % row["population"],
		"Neighbors: %s" % (", ".join(neighbors) if not neighbors.is_empty() else "—"),
	]
	if not court.is_empty():
		lines.append("Court: %s" % ", ".join(court))
	_panel_body.text = "\n".join(lines)

## Test seam.
func panel_title() -> Label:
	return _panel_title
```
