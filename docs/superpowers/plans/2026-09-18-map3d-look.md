# 3D Map Look Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the 3D MAP tab zoomable by trackpad and down to 800 m, sharp and natural on the ground, with rings for settlements, a scale bar and compass, drifting clouds with shadows, real rivers and a banded sea — at ≥ 60 fps.

**Architecture:** Ground quality is fixed in the DATA (`tools/map3d/bake_world.py` → Terrain3D's standard material), not a new terrain shader. Everything drawn over the ground is runtime code in `scripts/map3d/`: pure `RefCounted` maths classes (tested headless) plus thin nodes/shaders owned by `Map3DView`. A bench that drives the real `Map3DView` gates every task on frame time.

**Tech Stack:** Godot 4.7 GDScript (typed), GL Compatibility, Terrain3D 1.0.2, GUT; Python 3.14 + numpy 2.5 + Pillow 12 for the bake.

**Spec:** `docs/superpowers/specs/2026-09-18-map3d-look-design.md`

```gdscript
## The distance times a gesture's `factor`, clamped (a pinch is not a notch).
static func zoomed_by(distance: float, factor: float, min_d: float, max_d: float) -> float:
	return clampf(distance * factor, min_d, max_d)


## The rig's new target when the distance goes from `old_d` to `new_d` and
## the ground point `anchor` (the one under the pointer) must stay put: the
## view scales about the anchor, so every offset from it scales by new/old.
static func zoom_about(target: Vector2, anchor: Vector2, old_d: float, new_d: float) -> Vector2:
	if old_d <= 0.0:
		return target
	return anchor + (target - anchor) * (new_d / old_d)
```

```gdscript
## Layers the bench can switch off to price them, those that exist: the

func layer_names() -> PackedStringArray:
	var out := PackedStringArray()
	for n: String in ["Markers", "Lines", "Rivers", "Clouds"]:
		if _viewport.get_node_or_null(n) != null:
			out.append(n)
	if _world != null and _world.get_node_or_null("Sea") != null:
		out.append("Sea")
	if _world != null and _world.get_node_or_null("Sun") != null:
		out.append("Shadows")
	if get_node_or_null("Hud") != null:
		out.append("Hud")
	return out

func set_layer_visible(layer: String, on: bool) -> void:
	match layer:
		"Shadows":
			(_world.get_node("Sun") as DirectionalLight3D).shadow_enabled = on
		"Sea":
			(_world.get_node("Sea") as Node3D).visible = on
		"Hud":

		_:
			(_viewport.get_node(layer) as Node3D).visible = on
```

```gdscript
## Frame time of the 3D MAP as the player sees it (spec 2026-09-18 §5): the
## real GameRoot, the court, the 3D tab lit, the clock at 4x — at the near,
## mid and far zoom with every layer on, then each layer off in turn to price

const WARM := 30
const BUDGET_MS := 8.0
const STOPS := [["near", "map3d_zoom_min_m"], ["mid", "map3d_zoom_start_m"],
	["far", "map3d_zoom_max_m"]]


func _ready() -> void:
	DisplayServer.window_set_vsync_mode(DisplayServer.VSYNC_DISABLED)
	var size := Vector2i(2560, 1440)
	var args := OS.get_cmdline_user_args()
	var at := args.find("--size")
```

```glsl
// A settlement on the 3D map (placeholder, owner 2026-09-18): a hollow ring
// of constant SCREEN size, a billboard facing the camera, drawn without the
// depth test so it reads over a ridge like a pin. A QuadMesh of 1x1 is
// grown in the vertex stage to the ring's size at its distance (the
// px_per_dist sum border_line.gdshader uses); the fragment stage cuts an
// anti-aliased stroke with a faint dark keyline so a pale ring holds on
// pale ground.
shader_type spatial;
render_mode unshaded, cull_disabled, blend_mix, depth_draw_never, depth_test_disabled,
	shadows_disabled, fog_disabled;

uniform vec4 colour : source_color = vec4(1.0);
uniform vec4 keyline : source_color = vec4(0.08, 0.08, 0.08, 0.55);
uniform float ring_px = 14.0;  // outer diameter, window px
uniform float line_px = 2.0;   // stroke, window px
uniform float lift_px = 3.0;
uniform float px_per_dist = 0.001;

varying vec2 v_px;  // this fragment's offset from the centre, window px

void vertex() {
	vec3 centre = (MODEL_MATRIX * vec4(0.0, 0.0, 0.0, 1.0)).xyz;
	float m_per_px = length(centre - CAMERA_POSITION_WORLD) * px_per_dist;
```

```gdscript
## The yaw `t` (0..1) of the way home to north from `from_deg`, the short
## way round, smoothstepped so it starts and lands gently.
static func ease_yaw(from_deg: float, t: float) -> float:
	var start := wrapf(from_deg, -180.0, 180.0)
	if t <= 0.0:
		return from_deg
	return start * (1.0 - smoothstep(0.0, 1.0, clampf(t, 0.0, 1.0)))
```

```python
def coverage(land_loops, lake_loops, vw, vh, k, ss=4):
    """Share (0..1) of each vertex's square that is land: the loops filled at
    `ss` samples per vertex each way, lakes cut out, then averaged. Vertex v

def shore_heights(height, cov, sea_y, shore_m):
    """Land (cov 1) at least shore_m over the water plane, sea (cov 0) at
    least shore_m under it, and the fringe ramped through it so the coast
    crosses the plane where the loop does."""
    out = np.where(cov >= 1.0, np.maximum(height, sea_y + shore_m), height)
    out = np.where(cov <= 0.0, np.minimum(out, sea_y - shore_m), out)
    fringe = (cov > 0.0) & (cov < 1.0)
    return np.where(fringe, sea_y + shore_m * (2.0 * cov - 1.0), out).astype(np.float32)


def sample_bilinear(a, xs, ys):
    """a sampled at fractional (xs, ys), clamped to its edges."""
    h, w = a.shape
    xs = np.clip(xs, 0, w - 1)
    ys = np.clip(ys, 0, h - 1)
    x0 = np.minimum(np.floor(xs).astype(np.int64), w - 2)
    y0 = np.minimum(np.floor(ys).astype(np.int64), h - 2)
    fx, fy = xs - x0, ys - y0
    top = a[y0, x0] * (1 - fx) + a[y0, x0 + 1] * fx
    bot = a[y0 + 1, x0] * (1 - fx) + a[y0 + 1, x0 + 1] * fx
    return top * (1 - fy) + bot * fy


def blend_byte(w1, w2, dither):
    """Terrain3D's blend (0 base only .. 255 overlay only) from the top two
    weights: the overlay's share (0..0.5) through a smoothstep, so thin
    traces vanish and the crossing is steep, plus dither against banding."""
    r = np.where(w1 + w2 > 0, w2 / np.maximum(w1 + w2, 1e-6), 0.0)
    t = np.clip(r * 2.0 + dither, 0.0, 1.0)
    return np.round(255.0 * 0.5 * t * t * (3.0 - 2.0 * t)).astype(np.uint32)
```

```python
def resample_line(pts, step):
    """Points every `step` (same units) along a polyline, ends kept."""
    pts = np.asarray(pts, dtype=np.float64)
    seg = np.hypot(*np.diff(pts, axis=0).T)
    s = np.concatenate([[0.0], np.cumsum(seg)])
    if s[-1] <= 0:
        return pts[:1]
    t = np.append(np.arange(0.0, s[-1], step), s[-1])
    return np.column_stack([np.interp(t, s, pts[:, 0]), np.interp(t, s, pts[:, 1])])


def river_bed(ground, depth, drop, floor):
    """Bed heights source -> mouth: `depth` under the ground, but always at
    least `drop` below the step before, never under `floor`."""
    bed = np.empty(len(ground), dtype=np.float64)
    prev = np.inf
    for i, g in enumerate(ground):
        prev = max(min(g - depth, prev - drop), floor)
        bed[i] = prev
    return bed


def river_widths(n, w_mouth, gamma, w_source):
    """n widths growing from w_source to w_mouth as t^gamma (the 2D taper)."""
    t = np.linspace(0.0, 1.0, n) if n > 1 else np.ones(1)
    return np.maximum(w_mouth * t ** gamma, w_source)


def carve(height, wet, pts, bed, widths_v, rules):
    """Cut each sample's valley into `height` (vertex units, in place): flat
    bed across the water, banks rising river_bank_slope out to where they
    meet the ground; `wet` takes the band beside the water (0..1)."""
    H, W = height.shape
    slope_v = rules["river_bank_slope"] * VERTEX_M      # metres per vertex
    bank_max_v = rules["river_bank_max_m"] / VERTEX_M
    wet_v = rules["river_wet_m"] / VERTEX_M
    for (x, y), b, w in zip(pts, bed, widths_v):
        xi, yi = int(round(x)), int(round(y))
        g = height[min(max(yi, 0), H - 1), min(max(xi, 0), W - 1)]
        reach = w * 0.5 + min(max(g - b, 0.0) / slope_v + 1.0, bank_max_v)
        r = int(np.ceil(max(reach, w * 0.5 + wet_v))) + 1
        x0, x1 = max(xi - r, 0), min(xi + r + 1, W)
        y0, y1 = max(yi - r, 0), min(yi + r + 1, H)
        if x0 >= x1 or y0 >= y1:
            continue
        yy, xx = np.mgrid[y0:y1, x0:x1]
        d = np.hypot(xx - x, yy - y)
        prof = b + np.maximum(d - w * 0.5, 0.0) * slope_v
        sub = height[y0:y1, x0:x1]
        np.minimum(sub, np.where(d <= reach, prof, np.inf), out=sub)
        wsub = wet[y0:y1, x0:x1]
        np.maximum(wsub, np.clip(1.0 - (d - w * 0.5) / wet_v, 0.0, 1.0), out=wsub)
```

```gdscript
class_name Map3DRivers
extends RefCounted
## The 3D map's rivers as ONE ribbon mesh (pure, tested headless). The bake
## (tools/map3d/bake_world.py) carves each river's bed and writes its water
## line, source -> mouth, to rivers.json beside the built scene: per point
## [x_m, z_m, water_y_m, width_m, mouth 0..1] in world metres. Each point
## becomes two vertices ON the centreline whose NORMAL is the side to grow
## toward — water_river.gdshader widens them to the river's width, or to a
## minimum screen width far away, as border_line.gdshader does for borders.


## rivers.json text -> [{code, pos: PackedVector3Array, width, mouth:
## PackedFloat32Array}]; [] for anything unreadable.
static func parse(text: String) -> Array:
	var data: Variant = JSON.parse_string(text)
	if not (data is Dictionary) or not (data as Dictionary).has("rivers"):
		return []
	var out: Array = []
	for r: Variant in (data as Dictionary)["rivers"]:
		var pts: Array = (r as Dictionary).get("points", [])
		if pts.size() < 2:
			continue
		var pos := PackedVector3Array()
		var width := PackedFloat32Array()
		var mouth := PackedFloat32Array()
		for p: Array in pts:
			pos.append(Vector3(float(p[0]), float(p[2]), float(p[1])))
			width.append(float(p[3]))
			mouth.append(float(p[4]))
		out.append({"code": String((r as Dictionary).get("code", "")), "pos": pos,
			"width": width, "mouth": mouth})
	return out


## Mesh arrays for every course, or [] when there is none.
static func ribbon_arrays(courses: Array) -> Array:
	var verts := PackedVector3Array()
	var normals := PackedVector3Array()
	var uv := PackedVector2Array()
	var uv2 := PackedVector2Array()
	var index := PackedInt32Array()
	for c: Dictionary in courses:
		var pos: PackedVector3Array = c["pos"]
		var width: PackedFloat32Array = c["width"]
		var mouth: PackedFloat32Array = c["mouth"]
		var along := 0.0
		for i in pos.size():
			var a := pos[maxi(i - 1, 0)]
			var b := pos[mini(i + 1, pos.size() - 1)]
			var dir := Vector2(b.x - a.x, b.z - a.z).normalized()
			var side := Vector3(dir.y, 0.0, -dir.x)  # left of the flow
			if i > 0:
				along += Vector2(pos[i].x - pos[i - 1].x, pos[i].z - pos[i - 1].z).length()
			var base := verts.size()
			for s: int in [0, 1]:
				verts.append(pos[i])
				normals.append(side * (1.0 if s == 0 else -1.0))
				uv.append(Vector2(float(s), along))
				uv2.append(Vector2(width[i], mouth[i]))
			if i > 0:
				index.append_array([base - 2, base - 1, base, base - 1, base + 1, base])
	if verts.is_empty():
		return []
	var arrays: Array = []
	arrays.resize(Mesh.ARRAY_MAX)
	arrays[Mesh.ARRAY_VERTEX] = verts
	arrays[Mesh.ARRAY_NORMAL] = normals
	arrays[Mesh.ARRAY_TEX_UV] = uv
	arrays[Mesh.ARRAY_TEX_UV2] = uv2
	arrays[Mesh.ARRAY_INDEX] = index
	return arrays
```

```glsl
// River water on the 3D map (spec 2026-09-18 §7). Map3DRivers lays both
// vertices of a pair on the centreline; the vertex stage widens them to the
// river's width (UV2.x, metres) or to min_px on screen far away, and lifts
// them a little so a far river is never swallowed by the relief. The
// fragment stage scrolls two ripple normals DOWNSTREAM (UV.y is metres from
// the source), mixes in the sky by fresnel, shades the mouth (UV2.y) toward
// the sea's near band, and fades both edges so a river has no hard rim.
shader_type spatial;
render_mode blend_mix, depth_draw_never, cull_disabled, shadows_disabled, fog_disabled;

uniform vec4 water : source_color = vec4(0.5, 0.72, 0.87, 1.0);
uniform vec4 mouth_water : source_color = vec4(0.66, 0.77, 0.71, 1.0);
uniform vec4 sky : source_color = vec4(0.66, 0.75, 0.8, 1.0);
uniform sampler2D ripples : hint_normal, filter_linear_mipmap, repeat_enable;
uniform float ripple_m = 60.0;
uniform float flow_m_s = 0.8;
uniform float px_per_dist = 0.001;
uniform float min_px = 1.5;
uniform float lift_px = 1.0;
uniform float edge_soft = 0.35;
uniform float opacity = 0.92;

varying float v_across;
varying float v_along;
varying float v_width;
varying float v_mouth;

void vertex() {
	vec3 world = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz;
	float m_per_px = length(world - CAMERA_POSITION_WORLD) * px_per_dist;
	float half_w = max(0.5 * UV2.x, 0.5 * min_px * m_per_px);
	VERTEX += NORMAL * half_w;
	VERTEX.y += lift_px * m_per_px;
	NORMAL = vec3(0.0, 1.0, 0.0);
	v_across = UV.x;
	v_along = UV.y;
	v_width = 2.0 * half_w;
	v_mouth = UV2.y;
}

void fragment() {
	vec2 p = vec2((v_across - 0.5) * v_width, v_along) / ripple_m;
	vec3 n1 = texture(ripples, p + vec2(0.0, -TIME * flow_m_s / ripple_m)).xyz;
	vec3 n2 = texture(ripples, p * 1.7 + vec2(0.31, -TIME * flow_m_s * 1.3 / ripple_m)).xyz;
	NORMAL_MAP = normalize(n1 + n2 - 1.0) * 0.5 + 0.5;
	NORMAL_MAP_DEPTH = 0.6;
	float fres = pow(1.0 - clamp(dot(NORMAL, VIEW), 0.0, 1.0), 4.0);
	vec3 base = mix(water.rgb, mouth_water.rgb, v_mouth);
	ALBEDO = mix(base, sky.rgb, fres * 0.6);
	ROUGHNESS = 0.18;
	SPECULAR = 0.5;
	float edge = smoothstep(0.0, edge_soft, 1.0 - abs(v_across * 2.0 - 1.0));
	ALPHA = opacity * edge;
}
```

```gdscript
class_name Map3DCloudField
extends RefCounted
## Where the 3D map's clouds are (owner 2026-09-18: more than the 2D map's):
## a pure field the view steps every frame and draws. While the clock runs
## the sky fills at once over the view, clouds drift downwind at the clock's
## speed, drop out once well past the view and come in again upwind; when it
## stops they fade and are cleared. Look only: its own seeded generator,
## never the engine's stream. Positions are world XZ metres.

var clouds: Array = []  # {pos: Vector2, size: float, rot: float, life: float}
var fade := 0.0
var _rng := RandomNumberGenerator.new()
var _spawn_in := 0.0


func _init(seed_value: int) -> void:
	_rng.seed = seed_value


func step(delta: float, running: bool, clock_speed: int, view: Rect2, p: Dictionary) -> void:
	var fade_s := maxf(float(p["fade_s"]), 0.01)
	if not running:
		fade = maxf(fade - delta / fade_s, 0.0)
		if fade <= 0.0:
			clouds.clear()
			_spawn_in = 0.0
		return
	var first := clouds.is_empty() and fade <= 0.0
	fade = minf(fade + delta / fade_s, 1.0)
	var wind := Vector2.from_angle(deg_to_rad(float(p["wind_deg"])))
	var move := wind * float(p["speed_m_s"]) * float(maxi(clock_speed, 1)) * delta
	var keep: Array = []
	var margin := float(p["size_max"]) * 2.0
	for c: Dictionary in clouds:
		c["pos"] = (c["pos"] as Vector2) + move
		c["life"] = minf(float(c["life"]) + delta / fade_s, 1.0)
		if view.grow(margin).has_point(c["pos"]):
			keep.append(c)
	clouds = keep
	var cap := int(p["max"])
	if first:
		while clouds.size() < cap:
			clouds.append(_cloud(view, p, Vector2.ZERO))
		return
	_spawn_in -= delta
	if _spawn_in <= 0.0 and clouds.size() < cap:
		clouds.append(_cloud(view, p, wind))
		_spawn_in = float(p["interval_s"]) * _rng.randf_range(0.5, 1.5)


## A cloud over the view, or — with a wind — just off its upwind side.
func _cloud(view: Rect2, p: Dictionary, wind: Vector2) -> Dictionary:
	var pos := Vector2(_rng.randf_range(view.position.x, view.end.x),
		_rng.randf_range(view.position.y, view.end.y))
	var size := _rng.randf_range(float(p["size_min"]), float(p["size_max"]))
	if wind != Vector2.ZERO:
		pos -= wind * (view.size.length() * 0.5 + size)
	return {"pos": pos, "size": size, "rot": _rng.randf_range(0.0, TAU),
		"life": 1.0 if wind == Vector2.ZERO else 0.0}
```

```glsl
// The 3D map's clouds (owner 2026-09-18): flat quads high over the land,
// a soft noise alpha (CloudLayer.cloud_texture, the 2D clouds' own), lit by
// the sun so their undersides read, faded by the field (COLOR.a) and by
// nearness so a camera under the deck is never blinded. `shadow_pass` makes
// the twin that only casts: the same shape cut by alpha scissor.
shader_type spatial;
render_mode blend_mix, depth_draw_never, cull_disabled, diffuse_lambert, specular_disabled;

uniform sampler2D shape : source_color, filter_linear_mipmap, repeat_disable;
uniform float opacity = 0.6;
uniform float near_fade_m = 6000.0;
uniform bool shadow_pass = false;
uniform float shadow_cut = 0.35;

void fragment() {
	float a = texture(shape, UV).a * COLOR.a;
	if (shadow_pass) {
		ALBEDO = vec3(0.0);
		ALPHA = a;
		ALPHA_SCISSOR_THRESHOLD = shadow_cut;
	} else {
		float dist = length(VERTEX);
		ALBEDO = vec3(0.97);
		ALPHA = a * opacity * smoothstep(near_fade_m * 0.3, near_fade_m, dist);
	}
}
```

(A shadows-only instance never draws its colour, so the scissor branch only shapes the shadow. `render_mode` cannot switch per material, so `blend_mix` stays on both; if the shadow twin shows no shadow under Compatibility — check a shot at 3000 m — switch the twin to its own shader with `render_mode depth_draw_opaque` and the scissor, or take the spec's fallback: dark translucent ground quads offset along the sun.)

```gdscript
## The ground rect the camera sees (its four corners' ground hits), wide
## enough for the sky when a corner looks past the horizon.
func _visible_ground() -> Rect2:
	var r := Rect2(_target, Vector2.ZERO)
	for f: Vector2 in [Vector2.ZERO, Vector2(1, 0), Vector2(0, 1), Vector2.ONE]:
		var hit: Variant = ground_at(size * f)
		r = r.expand(Vector2((hit as Vector3).x, (hit as Vector3).z) if hit != null
			else _target + Vector2(_distance, _distance) * (f * 2.0 - Vector2.ONE) * 2.0)
	return r


func _step_clouds(delta: float) -> void:

		return

	var h := _repo.bal("map3d_cloud_height_m")
```
